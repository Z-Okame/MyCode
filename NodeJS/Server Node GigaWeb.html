function requestHandler(request, response) {
    const oggettoUrl = new URL(request.url, "http://localhost");
    const path = oggettoUrl.pathname;
    console.log(request.method, path);

    switch (path) {

readFile

Pagine HTML con fs.readFile

server.js, funzione inviaPagina

function inviaPagina(response, percorso) {
    fs.readFile(percorso, function (error, data) {
        if (error) {
            response.writeHead(500, { "Content-Type": "text/plain; charset=utf-8" });
            response.end("Errore interno del server");
        } else {
            response.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
            response.end(data);
        }
    });
}
  • callback (error, data): prima controlli l'errore, poi rispondi
  • errore → 500 con testo semplice; ok → 200 + text/html

Ricorda: readFile carica tutto il file in memoria e poi lo manda. Va bene per HTML piccoli.

stream

CSS e immagini con createReadStream + pipe

server.js, funzione inviaStream

function inviaStream(response, percorso, contentType) {
    const readable = fs.createReadStream(percorso);

    readable.on("open", () => {
        response.writeHead(200, { "Content-Type": contentType });
        readable.pipe(response);
    });

    readable.on("error", err => {
        console.error("Errore stream:", err.message);
        if (!response.headersSent) {
            response.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" });
        }
        response.end("File non trovato");
    });
}
  • open → il file esiste → solo ora scrivi gli header e fai partire la pipe
  • pipe(response) gestisce i chunk e chiude la risposta alla fine
  • error → 404; controlla headersSent prima di riscrivere gli header

Ricorda: il Content-Type cambia con il file: text/css, image/jpeg, text/html.

POST

Ricevere il corpo del POST

server.js, funzione riceviForm

function riceviForm(request, response) {
    let corpo = "";

    request.on("data", function (chunk) {
        corpo += chunk;
    });

    request.on("end", function () {
        const esito = valida(qs.parse(corpo));
        response.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
        response.end(esito.valido ? generaConferma(esito) : generaHtml(esito));
    });
}
  • il corpo arriva a pezzi: accumuli in corpo con data
  • solo in end hai tutto → qs.parse → valida
  • rispondi dentro end, mai prima

Ricorda: la chiave arriva dall'attributo name dell'input, non dall'id.

regex

Funzione valida(): un oggetto esito con un booleano per campo

server.js, regex in alto + funzione valida

const regexNome = /^[A-Za-zÀ-ÖØ-öø-ÿ' -]+$/;
const regexMatricola = /^\d{5}$/;
const regexEmail = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const regexTelefono = /^\+?\d{6,15}$/;
function valida(dati) {
    const cognome = leggi(dati, "cognome").trim();
    const nome = leggi(dati, "nome").trim();
    const matricola = leggi(dati, "matricola").trim();
    const regione = leggi(dati, "regione");
    const email = leggi(dati, "email").trim();
    const telefono = leggi(dati, "telefono").replace(/\s/g, "");
    const richieste = leggi(dati, "richieste");

    const esito = {
        cognome, nome, matricola, regione, email, telefono, richieste,
        cognomeOk: regexNome.test(cognome),
        nomeOk: regexNome.test(nome),
        matricolaOk: regexMatricola.test(matricola),
        regioneOk: regioni.includes(regione),
        contattoPresente: email !== "" || telefono !== "",
        emailOk: email === "" || regexEmail.test(email),
        telefonoOk: telefono === "" || regexTelefono.test(telefono)
    };

    esito.valido = esito.cognomeOk && esito.nomeOk && esito.matricolaOk &&
        esito.regioneOk && esito.contattoPresente && esito.emailOk && esito.telefonoOk;

    return esito;
}
  • .trim() prima del test: altrimenti " " (solo spazi) passa come nome valido
  • regione controllata con regioni.includes(), non solo "non vuota"
  • contatto: almeno uno tra email e telefono, e quelli compilati devono essere validi
  • leggi() evita il crash se arriva un campo ripetuto (array al posto di stringa)

Confronti

Domande tipiche di teoria: scegli e giustifica.

fs.readFile

Vantaggi

  • una callback sola, codice corto
  • sai se il file esiste prima di scrivere gli header

Svantaggi

  • file intero in RAM: male con file grandi
  • la risposta parte solo a lettura finita

createReadStream + pipe

Vantaggi

  • legge a chunk, memoria costante
  • pipe gestisce flusso e chiusura da sola

Svantaggi

  • errori via evento, header già partiti se fallisce a metà
  • più codice per un file di 2 KB

GET

Caratteristiche

  • dati nell'URL (query string): visibili, nella cronologia
  • lunghezza limitata, richiesta ripetibile
  • lettura: searchParams

POST

Caratteristiche

  • dati nel corpo: non compaiono nell'URL
  • arrivano a chunk → data + end
  • giusto per form con dati personali

Scelta di progetto: readFile per gli HTML (piccoli) e stream per CSS e immagini. La consegna non chiede stream: l'ho usato perché volevi vederli. Se il prof ti chiede solo readFile, sostituisci inviaStream con readFile e response.end(data): nel resto del codice non cambia nulla.

Errori da evitare

Trovati nei tuoi file allegati (bordo rosso) e limiti delle mie scelte (bordo arancio). Utile anche come lista "cosa controllo prima di consegnare".

  • Template string con apici singoli: non interpola nulla.In console.log('Server running at https.//${hostname}:${port}/') stampa le parole ${hostname} alla lettera, e il protocollo è scritto https.//. Servono i backtick e http://.Server_Url.js, riga 27
  • Tag non chiuso e MIME incompleto.<h1>...<h1> manca la / del tag di chiusura; la risposta HTML non dichiara text/html. Per JSON la convenzione è application/json minuscolo (con la J maiuscola in genere funziona, ma non è lo standard).Server_Url.js, righe 15 e 19
  • Switch senza default: richieste appese.Un path diverso da / e /recuperadati non riceve mai risposta, il browser gira a vuoto. Anche il &nbsp senza punto e virgola è scritto male.serverGetDinamico.js, righe 70-103 e 41, 51
  • Metodo non gestito → nessuna risposta.Se il metodo non è GET né POST (es. PUT), l'else if non risponde. Serve un ramo finale con 405.serverDinamico.js, righe 37-55
  • Regex con \s: " " risulta un nome valido./^[a-zA-Z\s]+$/ accetta una stringa di soli spazi. Soluzione: .trim() prima del test. Rifiuta anche accenti e apostrofi (D'Angelo, Sarà).serverGetDinamico.js riga 10, serverValidazione.js riga 17
  • Regione "valida" se non è vuota.regione !== "" accetta qualunque stringa inviata a mano (basta un POST con curl). Meglio controllare che sia nell'elenco.serverValidazione.js, riga 33
  • Valori rimessi nell'HTML senza escape.value="${esito.email}" e il testo della textarea finiscono nella pagina così come arrivano: un utente può inserire HTML o attributi. Qui lo risolve esc().serverValidazione.js, righe 131, 144, 148
  • Chunk scritti a mano.response.write(chunk) in un handler data non rispetta la contropressione; pipe sì ed è più corto.serverStaticoStream.js, righe 11-14
  • url.parse è API legacy.La documentazione Node la marca come deprecata a favore della WHATWG URL API (new URL(...)), che uso nel progetto. Funziona ancora, quindi se il prof insegna url.parse puoi usarlo senza paura.serverGetDinamico.js, riga 65 - fonte: nodejs.org/api/url.html
  • Il controllo non verifica la lunghezza massima.maxlength è solo nel browser: con una richiesta manuale un cognome di 500 caratteri passa. La consegna non lo chiede; se hai tempo aggiungi cognome.length <= 40.mio progetto, server.js
  • La regex dell'email è volutamente semplice.Accetta a@b.c. Per la verifica basta; la validità vera si prova solo mandando una mail.mio progetto, server.js
  • Il nome "GigaWeb" è una mia lettura della slide.Nella slide il titolo è poco leggibile. Le immagini sono segnaposto generati da me: sostituiscile con quelle della lezione tenendo gli stessi nomi (pic_trulli.jpg, img_chania.jpg) oppure aggiorna le route.mio progetto, Home.html

function inviaPagina(response, percorso) {
    fs.readFile(percorso, function (error, data) {
        if (error) {
            response.writeHead(500, { "Content-Type": "text/plain; charset=utf-8" });
            response.end("Errore interno del server");
        } else {
            response.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
            response.end(data);
        }
    });
}

function inviaStream(response, percorso, contentType) {
    const readable = fs.createReadStream(percorso);

    readable.on("open", () => {
        response.writeHead(200, { "Content-Type": contentType });
        readable.pipe(response);
    });

    readable.on("error", err => {
        console.error("Errore stream:", err.message);
        if (!response.headersSent) {
            response.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" });
        }
        response.end("File non trovato");
    });
}

function riceviForm(request, response) {
    let corpo = "";

    request.on("data", function (chunk) {
        corpo += chunk;
    });

    request.on("end", function () {
        const esito = valida(qs.parse(corpo));
        response.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
        response.end(esito.valido ? generaConferma(esito) : generaHtml(esito));
    });
}

function requestHandler(request, response) {
    const oggettoUrl = new URL(request.url, "http://localhost");
    const path = oggettoUrl.pathname;
    console.log(request.method, path);

    switch (path) {
        case "/":
        case "/Home.html":
            inviaPagina(response, "Home.html");
            break;

        case "/contatti.html":
            inviaPagina(response, "contatti.html");
            break;

        case "/stile":
            inviaStream(response, "./css/stile.css", "text/css; charset=utf-8");
            break;

        case "/immagini/pic_trulli.jpg":
            inviaStream(response, "./immagini/pic_trulli.jpg", "image/jpeg");
            break;

        case "/immagini/img_chania.jpg":
            inviaStream(response, "./immagini/img_chania.jpg", "image/jpeg");
            break;

        case "/recuperadati":
            if (request.method === "POST") {
                riceviForm(request, response);
            } else {
                response.writeHead(405, { "Content-Type": "text/plain; charset=utf-8", "Allow": "POST" });
                response.end("Metodo non consentito: usa POST");
            }
            break;

        default:
            response.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" });
            response.end("404 Not Found");
    }
}

const server = http.createServer(requestHandler);

server.listen(port, hostname, function () {
    console.log(`Server running at http://${hostname}:${port}/`);
});
